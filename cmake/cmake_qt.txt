# If the QT_DIR environment variable is set (e.g., to C:/Qt/6.9.3/mingw_64
# on Windows MSYS2), prepend it to CMAKE_PREFIX_PATH so find_package(Qt6)
# can locate it. Stored as a CACHE variable with FORCE so nested CMake
# invocations (subdirectories, ExternalProject, etc.) see the same value.
# Guarded with _QT_DIR_PROCESSED so the message prints once even though this
# file is included from many subdirectory CMakeLists.txt files.
if(DEFINED ENV{QT_DIR} AND NOT "$ENV{QT_DIR}" STREQUAL "" AND NOT _QT_DIR_PROCESSED)
    list(APPEND CMAKE_PREFIX_PATH "$ENV{QT_DIR}")
    set(CMAKE_PREFIX_PATH "${CMAKE_PREFIX_PATH}" CACHE STRING
        "Search prefixes for find_package" FORCE)
    set(_QT_DIR_PROCESSED 1 CACHE INTERNAL "")
    message(STATUS "QT_DIR=$ENV{QT_DIR} added to CMAKE_PREFIX_PATH")
endif()

if(NOT DEFINED QT_VERSION_MAJOR)
    find_package(Qt6 COMPONENTS Core Gui Widgets Xml Network OPTIONAL_COMPONENTS Test QUIET)
    if(Qt6_FOUND)
        set(QT_VERSION_MAJOR 6 CACHE INTERNAL "Qt major version in use")
        message(STATUS "Found Qt6 ${Qt6_VERSION}")
    else()
        find_package(Qt5 COMPONENTS Core Gui Widgets Xml Network REQUIRED)
        if(NOT Qt5Widgets_FOUND)
            message(STATUS "Failed to find Qt5Widgets required (on debian/ubuntu try 'sudo apt install qtbase5-dev')")
        elseif(NOT Qt5Gui_FOUND)
            message(STATUS "Failed to find Qt5Gui required (on debian/ubuntu try 'sudo apt install qtbase5-dev')")
        elseif(NOT Qt5Core_FOUND)
            message(STATUS "Failed to find Qt5Core required (on debian/ubuntu try 'sudo apt install qtbase5-dev')")
        elseif(NOT Qt5Xml_FOUND)
            message(STATUS "Failed to find Qt5Xml required (on debian/ubuntu try 'sudo apt install qtbase5-dev')")
        elseif(NOT Qt5Network_FOUND)
            message(STATUS "Failed to find Qt5Network required (on debian/ubuntu try 'sudo apt install qtbase5-dev')")
        endif()
        set(QT_VERSION_MAJOR 5 CACHE INTERNAL "Qt major version in use")
    endif()
endif()

# Create Qt5:: aliases pointing to Qt6:: targets so third-party CMakeLists
# files that hardcode Qt5:: continue to work when building with Qt6.
if(QT_VERSION_MAJOR EQUAL 6)
    foreach(_qt_mod Core Gui Widgets Xml Network Test)
        if(TARGET Qt6::${_qt_mod} AND NOT TARGET Qt5::${_qt_mod})
            add_library(Qt5::${_qt_mod} INTERFACE IMPORTED GLOBAL)
            set_target_properties(Qt5::${_qt_mod} PROPERTIES
                INTERFACE_LINK_LIBRARIES "Qt6::${_qt_mod}")
        endif()
    endforeach()
endif()
